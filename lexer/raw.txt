// raw_strings.dart
void main() {
  // В raw-строках escape-последовательности НЕ обрабатываются.
  String r1 = r"Line1\nLine2";     // буквально \n
  String r2 = r"Tab\there";        // буквально \t
  String r3 = r"Backslash\\here";  // буквально две обратных косых
  String r4 = r"Dollar $name";     // буквально $name — интерполяции нет
  String r5 = r"Quote \" here";    // буквально \" 
  String r6 = r'Quote \' here';    // буквально \'
  String r7 = r"Unicode \u0041";   // буквально \u0041
  String r8 = r"Extended \u{1F600}"; // буквально \u{1F600}

  // Raw-строка с реальными кавычками (через чередование)
  String r9  = r'He said "hi"';
  String r10 = r"She said 'hi'";
}
